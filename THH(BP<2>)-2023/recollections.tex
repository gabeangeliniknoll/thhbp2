\section{Recollections}
In this section, we recall the necessery results from~\cite{AKCH24}. 

\begin{proposition}\label{prop:rational}
There is an isomorphism of $\pi_{*}L_{\bQ}\BP\langle n\rangle=\bQ[v_{1},\dots ,v_{n}]$-algebras
\[ 
\pi_{*}L_{\bQ}\THH(\BP\langle n\rangle )\cong\bQ[v_1,\dots ,v_n]\langle \sigma v_1,\dots ,\sigma v_n\rangle \,.
\]
\end{proposition}
\begin{proof}
By~\cite[Proposition~3.7]{AKCH24}, we have
\[ 
    \pi_{*}\THH(\BP\langle n\rangle;\bQ )=\bQ\langle \sigma v_1,\dots ,\sigma v_n\rangle\,.
\]
We then observe that 
\[L_{\bQ}\THH(\BP\langle n\rangle )\simeq \THH(\BP\langle n\rangle;L_{\mathbb{Q}}\BP\langle n\rangle )\]
because $L_{\mathbb{Q}}$ is a smashing localization. 
We then consider the Atiyah--Hirzebruch spectral sequence 
\[
\THH_{*}(\BP\langle n\rangle;\mathbb{Q})\otimes_{\bQ} \pi_{*}L_{\mathbb{Q}}\BP\langle n\rangle\implies \THH_{*}(\BP\langle n\rangle;L_{\mathbb{Q}}\BP\langle n\rangle)
\]
associated to the multiplicative complete filtration $\tau_{\ge \bullet}L_{\mathbb{Q}}BP\langle n\rangle$ in $\mathbb{Q}$-modules~\cite[\S~3.2]{Hon20}. 
This spectral sequence has input 
\[ \mathbb{Q} \langle \sigma v_{1},\dots ,\sigma v_{n}\rangle \otimes_{\bQ}\pi_{*}L_{\bQ}\BP\langle n\rangle 
\]
by \cite[Proposition~3.7]{AKCH24}. It collapses because the targets of all differentials whose source is an indecomposable algebra generator land in zero groups. The abutment and the $\mr{E}_{\infty}$-term are free $\pi_{*}L_{\mathbb{Q}}\BP\langle n\rangle$-modules and consequently there are no $\pi_*L_{\bQ}\BP\langle n\rangle$-module extensions. 
There isn't room for algebra extensions. 
\end{proof}

\begin{proposition}
The groups 
\[ 
\THH_{s}(\BP\langle n\rangle )
\]
are finitely generated for all integers $s$. Moreover, for each $1\le j\le n$ we have 
\[
|\THH_{s}(\BP\langle n\rangle )|<\infty
\]
for each $s\ne \sum_{a=1}^n(2p^s-1)e_s+\sum_{b=1}^n(2p^b-2)k_b$ for integers $k_i\ge 0$ and integers $e_i\in \{0,1\}$ where $1\le i\le n$. 
\end{proposition}
\begin{proof}
Since $\pi_{a}\mathbb{S}_{(p)}$ and $\pi_{b}\BP\langle n\rangle_{(p)}$ are finitely generated $\mathbb{Z}_{(p)}$-modules for all integers $a,b$, the strongly convergent K\"unneth spectral sequence computing $\pi_{*}(\BP\langle n\rangle\wedge \BP\langle n\rangle)$ is finitely generated in each bidegree and has a vanishing line of postive slope so $\pi_{c}(\BP\langle n\rangle\wedge \BP\langle n\rangle)$ is finitely generated for each integer $c$. The same argument implies that $\THH_{s}(\BP\langle n\rangle)$ is finitely generated for each integer $s$. The second statement then follows from Proposition~\ref{prop:rational} and the classification of finitely generated $\mathbb{Z}_{(p)}$-modules. 
\end{proof}
